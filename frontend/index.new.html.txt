  msg.textContent = ''; btn.disabled = true;
  try {
    const res = await api('/api/leads', {
      method: 'POST',

    });
    msg.textContent = res.message;

  } catch (e) {
    msg.textContent = e.data && e.data.errors ? e.data.errors.map((x) => x.message).join(', ') : e.message;
  } finally { btn.disabled = false; }
